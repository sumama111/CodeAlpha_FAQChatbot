<script>
const faqs = [
  {q:"what is codealpha", a:"CodeAlpha AI Internship is a remote program offering hands-on experience in AI and machine learning."},
  {q:"how many tasks", a:"You need to complete 2-3 tasks to get the certificate."},
  {q:"what are perks", a:"Offer Letter, QR Certificate, ID Certificate, LOR, Job Support, Resume Help."},
  {q:"how to submit", a:"Upload to GitHub, post on LinkedIn, submit via WhatsApp form."},
  {q:"is it remote", a:"Yes, fully remote!"},
  {q:"duration", a:"1-3 months depending on your speed."},
  {q:"certificate", a:"Yes, QR-verified certificate after 2-3 tasks."},
  {q:"task 2", a:"Task 2 is a FAQ Chatbot using NLP and cosine similarity."},
  {q:"python", a:"Yes, Python is recommended for AI tasks."},
  {q:"nlp", a:"NLP is Natural Language Processing - AI that understands human language."}
];

function tokenize(t){return t.toLowerCase().replace(/[^\w\s]/g,'').split(/\s+/).filter(w=>w.length>0);}
function tf(tokens){const f={};tokens.forEach(t=>f[t]=(f[t]||0)+1);const n=tokens.length;for(let w in f)f[w]/=n;return f;}
function vocab(docs){const v=new Set();docs.forEach(d=>tokenize(d).forEach(t=>v.add(t)));return[...v];}
function vec(text,v){const t=tokenize(text),f=tf(t);return v.map(w=>f[w]||0);}
function cos(a,b){let d=0,na=0,nb=0;for(let i=0;i<a.length;i++){d+=a[i]*b[i];na+=a[i]*a[i];nb+=b[i]*b[i];}return(na&&nb)?d/Math.sqrt(na*nb):0;}

const V=vocab(faqs.map(f=>f.q));
const QV=faqs.map(f=>vec(f.q,V));

function getAnswer(input){
  const uv=vec(input,V);
  let bi=0,bs=-1;
  QV.forEach((q,i)=>{const s=cos(uv,q);if(s>bs){bs=s;bi=i;}});
  return(bs>0.1)?faqs[bi].a:"Sorry, ask about CodeAlpha internship, tasks, or perks.";
}

function addMsg(text,isUser){
  const d=document.getElementById('messages');
  const m=document.createElement('div');
  m.className=isUser?'user-msg':'bot-msg';
  m.textContent=(isUser?'👤 ':'🤖 ')+text;
  d.appendChild(m);
  d.scrollTop=d.scrollHeight;
}

function send(){
  const i=document.getElementById('userInput');
  const t=i.value.trim();
  if(!t)return;
  addMsg(t,true);
  i.value='';
  setTimeout(()=>addMsg(getAnswer(t),false),500);
}

function ask(q){
  document.getElementById('userInput').value=q;
  send();
}
